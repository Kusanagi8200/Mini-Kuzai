```python
import concurrent.futures
import urllib.request
import re

def get_html_title(url):
    try:
        with urllib.request.urlopen(url) as response:
            html = response.read()
            match = re.search(r&#x27;&lt;title.*?&gt;(.*)&lt;/title&gt;&#x27;, html.decode(&#x27;utf-8&#x27;))
            if match:
                return match.group(1)
    except Exception as e:
        print(f&quot;Error fetching HTML title for {url}: {e}&quot;)
        return None

def get_html_titles(urls):
    with concurrent.futures.ThreadPoolExecutor() as executor:
        results = executor.map(get_html_title, urls)
    return list(results)
```
