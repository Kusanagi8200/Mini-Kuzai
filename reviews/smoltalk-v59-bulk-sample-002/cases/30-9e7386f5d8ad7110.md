```python
def approximate_logarithm(x, epsilon):
    num_trapezoids = int(1 / epsilon)
    dx = (x - 1) / num_trapezoids
    log_x = 0
    left_x = 1
    for _ in range(num_trapezoids):
        left_height = 1 / left_x
        right_height = 1 / (left_x + dx)
        log_x += dx * (left_height + right_height) / 2
        left_x += dx
    return log_x
```
