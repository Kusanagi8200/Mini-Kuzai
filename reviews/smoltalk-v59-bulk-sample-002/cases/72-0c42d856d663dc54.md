### Turn 6 - ASSISTANT

<pre>
SELECT E.FirstName, E.LastName, E.Age, E.Salary,
       A.AverageSalary * 0.5 AS MinSalary,
       A.AverageSalary * 2 AS MaxSalary
FROM Employees E
JOIN (
    SELECT Age, AVG(Salary) AS AverageSalary
    FROM Employees
    GROUP BY Age
) A ON E.Age = A.Age
WHERE E.Salary &lt; A.AverageSalary * 0.5 OR E.Salary &gt; A.AverageSalary * 2;
</pre>
